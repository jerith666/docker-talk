# cmd line script

#
# -- the socat server, inside and outside of docker
#

# create the socat static binary
pygmentize socat-static.nix
nix-build socat-static.nix -o socat-static; cp -v socat-static/bin/socat socat
file socat

# demonstrate the simple server outside docker:
./socat TCP-LISTEN:1234 OPEN:message-of-the-day.txt
telnet 127.0.0.1 1234

# demonstrate building a container with the simple server in it:
# note the 'FROM scratch'
pygmentize Dockerfile
docker build --progress plain .

docker run --interactive --tty --rm --publish 1234:1234 $( docker build --quiet . )
telnet 127.0.0.1 1234

#
# -- the docker container as an OS process
#

# demonstrate that the running "container" is merely a process
docker run --interactive --tty --rm --publish 1234:1234 $( docker build --quiet . )
docker ps
ps aux -w -w | grep /socat

# show what's special about the container process
su -l
cd /proc/<pid>
ls -alF

# show that it has its own root filesystem
grep ' / ' mounts
ls -alF /
ls -alF root/
# and that root filesystem restricts what executables it can see
nsenter -t $(basename $(pwd)) --mount ls /

# show that it has its own networking (but tied to the host via the 172.17.0.0/16 subnet)
ip address
nsenter -t $(basename $(pwd)) --net ip address
ip route
nsenter -t $(basename $(pwd)) --net ip route

# show that it has its own hostname
hostname
nsenter -t $(basename $(pwd)) --uts hostname

# show how it can be resource-limited
cat cgroup
head /sys/fs/cgroup/system.slice/docker-*.scope/pids.*

docker run --interactive --tty --rm --publish 1234:1234 --pids-limit 3 $( docker build --quiet . )
cd /proc/<pid>
cat cgroup
head /sys/fs/cgroup/system.slice/docker-*.scope/pids.*
telnet 127.0.0.1 1234 # works
telnet 127.0.0.1 1234 # works
telnet 127.0.0.1 1234 # boom

#
# -- a deeper look into the container filesystem, and what an image is
#

docker run --interactive --tty --rm --publish 1234:1234 $( docker build --quiet . )
cd /proc/<pid>
grep ' / ' mounts
# discuss overlay mounts
docker image inspect $( docker build --quiet . )

grep ' / ' mounts | cut -d ' ' -f 4 | cut -d , -f 3 | cut -d = -f 2 | tr : '\n' | while read layer; do echo; echo; echo $layer is a symlink:; ls -alF $layer; echo; echo inside the ${layer}/ symlink is:; ls -alF ${layer}/; done

# export the image
docker image save $( docker build --quiet . ) -o saved-image
file saved-image

# look inside it
mkdir tmp
cd tmp
tar xvf ../saved-image

# the older docker manifest
cat manifest.json | jq

# show the config
cat $( cat manifest.json | jq '.[0].Config' -r ) | jq

# show the layer containing 'socat'
file $( cat manifest.json | jq '.[0].Layers[0]' -r )
tar tvf $( cat manifest.json | jq '.[0].Layers[0]' -r )

# show the layer containing message-of-the-day.txt
file $( cat manifest.json | jq '.[0].Layers[1]' -r )
tar tvf $( cat manifest.json | jq '.[0].Layers[1]' -r )

# the newer OCI index compared to the old docker manifest:
cat index.json | jq
cat blobs/sha256/$(cat index.json | jq '.manifests[0].digest | split (":") | .[1]' -r) | jq
# both reference the same config and layers

#
# -- using nix to make an image
#

pygmentize image.nix
nix-build image.nix
file result
tar tzvf result

# look inside the nix-created image
mkdir nix-tmp
cd nix-tmp
tar xzvf ../result

# there is only an older docker manifest
cat manifest.json | jq

# show the config
cat $( cat manifest.json | jq '.[0].Config' -r ) | jq

# show the layer
file $( cat manifest.json | jq '.[0].Layers[0]' -r )
tar tvf $( cat manifest.json | jq '.[0].Layers[0]' -r )

# nix has a second, basically empty, layer
file $( cat manifest.json | jq '.[0].Layers[1]' -r )
tar tvf $( cat manifest.json | jq '.[0].Layers[1]' -r )

#
# -- load the nix-built image into docker and run it
#

cd ..
docker load < result

docker image inspect <that-id> | jq

docker run --interactive --tty --rm --publish 1234:1234 <that-id>
telnet 127.0.0.1 1234

# inspect the running container a bit
docker run --interactive --tty --rm --publish 1234:1234 <that-id>

# the filesystem as seen by the container process
ps aux -w -w | grep /socat
su -l
cd /proc/<pid>
ls -alF
ls -alF root/
tree root/nix/
